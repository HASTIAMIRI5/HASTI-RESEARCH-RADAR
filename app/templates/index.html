from __future__ import annotations import json import sqlite3 from pathlib import Path from typing import Any DB_DIR = Path(__file__).resolve().parents[1] / "data" DB_PATH = DB_DIR / "research_briefs.db" def init_db() -> None: DB_DIR.mkdir(parents=True, exist_ok=True) with sqlite3.connect(DB_PATH) as conn: conn.execute( """ CREATE TABLE IF NOT EXISTS briefs ( id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT NOT NULL, topic TEXT NOT NULL, goal TEXT, audience TEXT, source_mix TEXT, brief_json TEXT NOT NULL, created_at TEXT DEFAULT CURRENT_TIMESTAMP ) """ ) conn.commit() def save_brief(payload: dict[str, Any]) -> int: init_db() with sqlite3.connect(DB_PATH) as conn: cursor = conn.execute( """ INSERT INTO briefs (title, topic, goal, audience, source_mix, brief_json) VALUES (?, ?, ?, ?, ?, ?) """, ( payload["title"], payload["topic"], payload.get("goal", ""), payload.get("audience", ""), payload.get("source_mix", ""), payload["brief"], ), ) conn.commit() return int(cursor.lastrowid) def list_recent_briefs(limit: int = 10) -> list[dict[str, Any]]: init_db() with sqlite3.connect(DB_PATH) as conn: rows = conn.execute( """ SELECT id, title, topic, goal, audience, source_mix, brief_json, created_at FROM briefs ORDER BY created_at DESC LIMIT ? """, (limit,), ).fetchall() briefs: list[dict[str, Any]] = [] for row in rows: brief_id, title, topic, goal, audience, source_mix, brief_json, created_at = row brief_data = json.loads(brief_json) brief_data["id"] = brief_id brief_data["title"] = title brief_data["goal"] = goal brief_data["audience"] = audience brief_data["source_mix"] = source_mix brief_data["created_at"] = created_at briefs.append(brief_data) return briefs def get_brief(brief_id: int) -> dict[str, Any] | None: init_db() with sqlite3.connect(DB_PATH) as conn: row = conn.execute( """ SELECT id, title, topic, goal, audience, source_mix, brief_json, created_at FROM briefs WHERE id = ? """, (brief_id,), ).fetchone() if row is None: return None _id, title, topic, goal, audience, source_mix, brief_json, created_at = row brief_data = json.loads(brief_json) brief_data["id"] = _id brief_data["title"] = title brief_data["topic"] = topic brief_data["goal"] = goal brief_data["audience"] = audience brief_data["source_mix"] = source_mix brief_data["created_at"] = created_at return brief_data
